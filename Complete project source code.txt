import java.util.Scanner;

class Person {
    String name;
    int age;

    Person(String name, int age) {
        this.name = name;
        this.age = age;
    }
}

class Patient extends Person {
    int id;
    String disease;

    Patient(int id, String name, int age, String disease) {
        super(name, age);
        this.id = id;
        this.disease = disease;
    }

    void display() {
        System.out.println("\nPatient ID: " + id);
        System.out.println("Name: " + name);
        System.out.println("Age: " + age);
        System.out.println("Disease: " + disease);
    }
}

public class ClinicManager {

    static void appointment(String doctor, String date) {
        System.out.println("Doctor: " + doctor);
        System.out.println("Date: " + date);
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Patient ID: ");
        int id = sc.nextInt();
        sc.nextLine();

        System.out.print("Patient Name: ");
        String name = sc.nextLine();

        System.out.print("Age: ");
        int age = sc.nextInt();
        sc.nextLine();

        System.out.print("Disease: ");
        String disease = sc.nextLine();

        Patient p = new Patient(id, name, age, disease);

        System.out.print("Doctor: ");
        String doctor = sc.nextLine();

        System.out.print("Appointment Date: ");
        String date = sc.nextLine();

        String[] medicine = new String[2];

        System.out.print("Medicine 1: ");
        medicine[0] = sc.nextLine();

        System.out.print("Medicine 2: ");
        medicine[1] = sc.nextLine();

        System.out.println("\n--- CLINIC DETAILS ---");
        p.display();

        System.out.println("\n--- APPOINTMENT ---");
        appointment(doctor, date);

        System.out.println("\n--- PRESCRIPTION ---");
        for (String m : medicine)
            System.out.println(m);

        if (age >= 18)
            System.out.println("Category: Adult");
        else
            System.out.println("Category: Child");

        sc.close();
    }
}
